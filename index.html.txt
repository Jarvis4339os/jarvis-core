  <script>
    (function () {
      var cv = document.getElementById('orbTex');
      if (!cv || !cv.getContext) return;
      var S = 256, ctx = cv.getContext('2d');
      cv.width = S; cv.height = S;
      var img = ctx.createImageData(S, S), d = img.data;
      var G = 96, grid = new Float32Array(G * G), seed = 987654321;
      var rnd = function () { seed = (seed * 1103515245 + 12345) & 0x7fffffff; return seed / 0x7fffffff; };
      for (var i = 0; i < grid.length; i++) grid[i] = rnd();
      var gv = function (x, y) { return grid[(y & (G - 1)) * G + (x & (G - 1))]; };
      var sm = function (t) { return t * t * (3 - 2 * t); };
      function nz(x, y) {
        var x0 = Math.floor(x), y0 = Math.floor(y), fx = sm(x - x0), fy = sm(y - y0);
        var a = gv(x0, y0), b = gv(x0 + 1, y0), c = gv(x0, y0 + 1), e = gv(x0 + 1, y0 + 1);
        var top = a + (b - a) * fx, bot = c + (e - c) * fx;
        return top + (bot - top) * fy;
      }
      function fbm(x, y) {
        return nz(x, y) * 0.52 + nz(x * 2.3, y * 2.3) * 0.26 + nz(x * 5.3, y * 5.3) * 0.14 + nz(x * 11.1, y * 11.1) * 0.08;
      }
      for (var y = 0; y < S; y++) {
        for (var x = 0; x < S; x++) {
          var o = (y * S + x) * 4;
          var u = (x / S) * 2 - 1, v = (y / S) * 2 - 1, r2 = u * u + v * v;
          if (r2 > 1) { d[o + 3] = 0; continue; }
          var z = Math.sqrt(1 - r2);
          var lon = Math.atan2(u, z), lat = Math.asin(v);
          var n1 = fbm(lon * 3.6 + 12.5, lat * 3.6 + 7.25);
          var n2 = fbm(lon * 9.2 + 3.1, lat * 9.2 + 5.7);
          var lit = Math.max(0, u * -0.42 + v * -0.5 + z * 0.76);
          var t = n1 * 0.62 + n2 * 0.2 + lit * 0.4;
          var a = Math.min(1, Math.abs(t - 0.43) * 1.6) * 0.6;
          var val = t > 0.43 ? 255 : 0;
          d[o] = val; d[o + 1] = val; d[o + 2] = val;
          d[o + 3] = Math.round(a * 255 * (0.32 + z * 0.68));
        }
      }
      ctx.putImageData(img, 0, 0);
    })();
  </script>
